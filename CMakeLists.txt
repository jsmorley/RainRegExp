cmake_minimum_required(VERSION 3.21)
project(RainRegExp VERSION 5.1 LANGUAGES C CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
file(GLOB PCRE_SOURCES CONFIGURE_DEPENDS third_party/pcre/pcre16_*.c)
add_library(rainmeter_pcre STATIC ${PCRE_SOURCES})
target_include_directories(rainmeter_pcre PUBLIC third_party/pcre)
target_compile_definitions(rainmeter_pcre PUBLIC PCRE_STATIC PRIVATE HAVE_CONFIG_H _CRT_SECURE_NO_WARNINGS)
add_library(rainregexp_core STATIC src/Core.cpp src/Network.cpp src/Appearance.cpp)
target_include_directories(rainregexp_core PUBLIC src)
target_compile_definitions(rainregexp_core PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(rainregexp_core PUBLIC rainmeter_pcre wininet)
add_executable(RainRegExp WIN32 src/Main.cpp src/RainRegExp.rc)
target_link_libraries(RainRegExp PRIVATE rainregexp_core comctl32 comdlg32 wininet shell32 advapi32 gdiplus shlwapi uxtheme dwmapi)
if(MSVC)
  target_link_options(RainRegExp PRIVATE /MANIFEST:NO)
  target_compile_options(rainregexp_core PRIVATE /W4 /utf-8)
  target_compile_options(RainRegExp PRIVATE /W4 /utf-8)
endif()
add_custom_command(TARGET RainRegExp POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/RainRegExp.ini" "$<TARGET_FILE_DIR:RainRegExp>/RainRegExp.ini")
include(CTest)
if(BUILD_TESTING)
  add_executable(CoreTests tests/CoreTests.cpp)
  target_link_libraries(CoreTests PRIVATE rainregexp_core)
  if(MSVC)
    target_compile_options(CoreTests PRIVATE /W4 /utf-8)
  endif()
  add_test(NAME CoreTests COMMAND CoreTests)
  add_executable(NetworkTests tests/NetworkTests.cpp)
  target_link_libraries(NetworkTests PRIVATE rainregexp_core ws2_32)
  if(MSVC)
    target_compile_options(NetworkTests PRIVATE /W4 /utf-8)
  endif()
  add_test(NAME NetworkTests COMMAND NetworkTests)
  set_tests_properties(NetworkTests PROPERTIES TIMEOUT 45)
endif()
